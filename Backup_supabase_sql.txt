-- 1. Criação da Tabela de Despesas
CREATE TABLE IF NOT EXISTS public.despesas (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
    valor numeric NOT NULL,
    categoria text NOT NULL,
    data date NOT NULL,
    forma_pagamento text,
    pagamento text, -- Mantido por compatibilidade histórica
    descricao text,
    comprovante_url text
);

-- Remove restrição NOT NULL caso exista em colunas antigas
ALTER TABLE public.despesas ALTER COLUMN pagamento DROP NOT NULL;

-- Desativa RLS da tabela despesas para permitir leituras e inserções públicas/anônimas
ALTER TABLE public.despesas DISABLE ROW LEVEL SECURITY;


-- 2. Configuração do Storage (Bucket de Comprovantes)
INSERT INTO storage.buckets (id, name, public) 
VALUES ('comprovantes', 'comprovantes', true)
ON CONFLICT (id) DO UPDATE SET public = true;

-- Concede permissões no schema de Storage para o perfil anon e public
GRANT ALL ON SCHEMA storage TO anon, authenticated, postgres, service_role;
GRANT ALL ON ALL TABLES IN SCHEMA storage TO anon, authenticated, postgres, service_role;
GRANT ALL ON ALL SEQUENCES IN SCHEMA storage TO anon, authenticated, postgres, service_role;

-- Limpeza de políticas antigas
DROP POLICY IF EXISTS "permitir_todos_comprovantes" ON storage.objects;
DROP POLICY IF EXISTS "permitir_upload_anon_comprovantes" ON storage.objects;
DROP POLICY IF EXISTS "permitir_select_anon_comprovantes" ON storage.objects;

-- Política universal de acesso ao bucket 'comprovantes'
CREATE POLICY "permitir_todos_comprovantes"
ON storage.objects
FOR ALL
TO anon, authenticated, public
USING (bucket_id = 'comprovantes')
WITH CHECK (bucket_id = 'comprovantes');

-- Atualização do cache de esquema da API do Supabase
NOTIFY pgrst, 'reload schema';
